#!/usr/bin/python3
"""Tests for BaseGeometry class."""

BaseGeometry = __import__('4-base_geometry').BaseGeometry

bg = BaseGeometry()

# Test area() method
try:
    bg.area()
except Exception as e:
    print(e)  # Should print: area() is not implemented

# Test integer_validator() method

# Valid value
bg.integer_validator("width", 10)
bg.integer_validator("height", 1)

# Invalid type
try:
    bg.integer_validator("width", "10")
except Exception as e:
    print(e)  # Should print: width must be an integer

try:
    bg.integer_validator("height", 3.5)
except Exception as e:
    print(e)  # Should print: height must be an integer

# Boolean should fail
try:
    bg.integer_validator("width", True)
except Exception as e:
    print(e)  # Should print: width must be an integer

# Invalid value (<=0)
try:
    bg.integer_validator("height", 0)
except Exception as e:
    print(e)  # Should print: height must be greater than 0

try:
    bg.integer_validator("length", -5)
except Exception as e:
    print(e)  # Should print: length must be greater than 0

print("All tests executed.")
